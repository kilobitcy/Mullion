/// 六个框共用的外壳。F277 起换成 `egui::Modal`:带变暗遮罩、挡下层点击 ——
/// 「文件已存在」这类框此前是普通居中 Window,淹在终端画面里(实报)。

/// F203 时代这个 `true` 来自标题栏 ✕;语义不变,来源换了。
fn modal<R>(ctx: &egui::Context, title: &str, body: impl FnOnce(&mut egui::Ui) -> R) -> bool {
    let resp = egui::Modal::new(egui::Id::new(("files_dialog", title))).show(ctx, |ui| {
        crate::ui::annotate::mark(ui.ctx(), format!("{title}对话框"), ui.max_rect());
        ui.heading(title);
        ui.separator();
        body(ui)
    });
    resp.should_close()
}

        }
    }
}

```rust
pub fn selection_text(&self) -> Option<String> {
    let raw = self.term.selection_to_string().filter(|s| !s.is_empty())?;
    Some(self.join_hard_wraps(raw))
}

/// F284:tmux 重绘走逐行显式定位,WRAPLINE 打不上,上游把软换行当真换行。
/// 启发式:选区里某物理行**最后一列非空**(含宽字符 spacer)且下一行仍在
/// 选区内 → 该处的 `\n` 是 tmux 折行,去掉。已知误伤(用户拍板接受):
/// 恰好排满整行宽、以真换行结尾的行会被拼进下一行,含 TUI 全宽边框。
///
/// 映射关系:上游只在「无 WRAPLINE 的行尾」补 `\n`,所以 raw 里第 k 个
/// `\n` 对应选区内第 k 个无 WRAPLINE 的行边界。数量对不上(防御:块选区
/// 等奇形)就原样返回,绝不按错位的映射改字符串。
fn join_hard_wraps(&self, raw: String) -> String {
    /* 1. 取选区范围 self.term.selection.as_ref().and_then(|s| s.to_range(&self.term));
       取不到 → 原样返回。
       2. 遍历 range.start.line ..= range.end.line-1,收集每个「无 WRAPLINE」
       行的「最后一列是否被占」(cell.c != ' ' 或带宽字符 spacer flag ——
       两种 spacer 见 docs/gui-render-gotchas.md)。
       3. 若该列表长度 != raw 里 '\n' 个数 → 原样返回。
       4. 按位置重组:该边界「最后一列被占」→ 不写 '\n',否则照写。 */
}
```
