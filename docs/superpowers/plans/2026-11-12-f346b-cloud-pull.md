```rust
    /// F346-b:取一个对象的全部字节。**只读**,不在 `write` feature 后面。
    ///
    /// `max_bytes` 是硬上限:超过即 `TooLarge`,不会把整份读进内存。ureq 的
    /// `read_to_vec()` 不带 `.limit()` 时默认 10 MB 上限 —— 必须显式给,否则
    /// 上限由库的默认值决定,而不是调用方。
    pub fn get(&self, key: &str, amz_date: &str, max_bytes: u64) -> Result<Vec<u8>, CloudError> {
        let t = self.endpoint.target(key);
        let payload = sha256_hex(b"");
        let auth = sigv4::authorization(
            &sigv4::Request {
                method: "GET",
                host: &t.host,
                path: &t.path,
                query: "",
                payload_sha256: &payload,
                amz_date,
                extra_headers: &[],
            },
            &self.creds.access_key_id,
            &self.creds.secret_access_key,
            &self.region,
            "s3",
        );
        let mut resp = self
            .agent
            .get(&t.url)
            .header("authorization", &auth)
            .header("x-amz-date", amz_date)
            .header("x-amz-content-sha256", &payload)
            .call()
            .map_err(transport)?;
        let code = resp.status().as_u16();
        if (300..400).contains(&code) {
            return Err(redirect_err(code, &resp));
        }
        if !(200..300).contains(&code) {
            return Err(status_err(code, resp));
        }
        match resp.body_mut().with_config().limit(max_bytes).read_to_vec() {
            Ok(v) => Ok(v),
            Err(ureq::Error::BodyExceedsLimit(_)) => Err(CloudError::TooLarge { limit: max_bytes }),
            Err(e) => Err(transport(e)),
        }
    }
```

若 `ureq::Error::BodyExceedsLimit` 在该路径下被包成 `ureq::Error::Io(..)`(`a_get_larger_than_the_limit_is_refused` 红),读

```rust
/// F346-b:「最新那份」的键 = 认得出的键里序号最大的那条(与 [`next_seq`] 同口径)。
/// 一条都认不出 → `None`(空桶,或前缀填错,由调用方按列表是否为空分开报)。
pub fn latest_key(prefix: &str, keys: &[String]) -> Option<String> {
    keys.iter()
        .filter_map(|k| parse_seq(prefix, k).map(|s| (s, k)))
        .max_by_key(|(s, _)| *s)
        .map(|(_, k)| k.clone())
}

/// F346-b:`cloud.toml` 存在但读不懂(非 UTF-8 / toml 解析失败)。缺文件不算。
///
/// [`load`] 对这两种情况都给默认值、把标记藏在私有字段里 —— 桌面靠 `save`
/// 拒写就够了,但手机界面要**事先**区分「损坏」与「没填」(设计 R1)。
pub fn is_corrupt(dir: &Path) -> bool {
    let path = dir.join(CLOUD_FILE);
    if !path.exists() {
        return false;
    }
    match std::fs::read_to_string(&path) {
        Ok(text) => toml::from_str::<CloudConfig>(&text).is_err(),
        Err(_) => true,
    }
}

/// F346-b:用 `key_source` 的密钥封一段本机字节,**不需要打开 `Vault`**。
///
/// 与 [`crate::vault::Vault::seal_local`] 在 Keyring 方案下字节同构(有测试钉住)。
/// 手机端要用它:`vault/` 坏了时拉取是唯一的自愈手段,解 SK 不能先要求库是好的(设计 D3)。
/// 同 `seal_local` 的红线:产出**不许**进任何离开本机的东西。
fn seal_with(key_source: &dyn MasterKeySource, plain: &[u8]) -> Result<Vec<u8>, StoreError> {
    let payload = crate::crypto::encrypt(&key_source.load_or_create()?, plain)?;
    Ok(crate::secrets_file::encode(
        &crate::secrets_file::Scheme::Keyring,
        &payload,
    ))
}

fn open_with(key_source: &dyn MasterKeySource, sealed_b64: &str, what: &str) -> Result<String, StoreError> {
    if sealed_b64.is_empty() {
        return Ok(String::new());
    }
    let blob = b64()
        .decode(sealed_b64)
        .map_err(|e| StoreError::CorruptSecrets(format!("cloud.toml 的{what}密文不是合法 base64:{e}")))?;
    let (_, payload) = crate::secrets_file::parse(&blob)?;
    let plain = crate::crypto::decrypt(&key_source.load_or_create()?, payload)?;
    String::from_utf8(plain).map_err(StoreError::from)
}

/// [`set_secret_key`] 的免 `Vault` 版(F346-b)。
pub fn set_secret_key_with(
    cfg: &mut CloudConfig,
    key_source: &dyn MasterKeySource,
    sk: &str,
) -> Result<(), StoreError> {
    cfg.secret_sealed = b64().encode(seal_with(key_source, sk.as_bytes())?);
    Ok(())
}

/// [`secret_key`] 的免 `Vault` 版(F346-b)。空 = 没填过。
pub fn secret_key_with(cfg: &CloudConfig, key_source: &dyn MasterKeySource) -> Result<String, StoreError> {
    open_with(key_source, &cfg.secret_sealed, "")
}

/// [`set_passphrase`] 的免 `Vault` 版(F346-b)。空口令拒绝,理由同原版。
pub fn set_passphrase_with(
    cfg: &mut CloudConfig,
    key_source: &dyn MasterKeySource,
    pass: &str,
) -> Result<(), StoreError> {
    if pass.is_empty() {
        return Err(StoreError::Kdf("备份口令不能为空".into()));
    }
    cfg.passphrase_sealed = b64().encode(seal_with(key_source, pass.as_bytes())?);
    Ok(())
}

/// [`passphrase`] 的免 `Vault` 版(F346-b)。空 = 没设过。
pub fn passphrase_with(cfg: &CloudConfig, key_source: &dyn MasterKeySource) -> Result<String, StoreError> {
    open_with(key_source, &cfg.passphrase_sealed, "口令")
}
```

```rust
//! F346-b:只回放 LIST / GET 的假 S3(path-style,bucket `b`)。`tests/cloud_pull.rs`
//! 与 `examples/fake_s3_pack.rs` 用 `#[path]` 共用。**不验签**(签名对不对是
//! mullion-cloud `sigv4.rs` 官方向量的活)。
#![allow(dead_code)]

use std::collections::BTreeMap;
use std::io::{BufRead, BufReader, Write};
use std::net::{TcpListener, TcpStream};
use std::sync::{Arc, Mutex};
use std::thread;

#[derive(Clone, Default)]
pub struct FakeS3 {
    /// 键 → 对象字节。LIST 列出全部键(前缀匹配),GET 按键取。
    pub objects: BTreeMap<String, Vec<u8>>,
    /// 设了就对 GET 一律回这个(状态码,正文),用来测 404/403。
    pub get_override: Option<(u16, String)>,
}

pub struct Running {
    pub port: u16,
    /// 收到的请求行(`GET /b/...`),测试断言「没发过写请求」用。
    pub seen: Arc<Mutex<Vec<String>>>,
}

impl FakeS3 {
    pub fn with_object(key: &str, body: Vec<u8>) -> Self {
        let mut objects = BTreeMap::new();
        objects.insert(key.to_string(), body);
        Self { objects, get_override: None }
    }

    /// 后台线程里一直服务到进程结束。`addr` 例如 `"127.0.0.1:0"`。
    pub fn spawn(self, addr: &str) -> Running {
        let listener = TcpListener::bind(addr).expect("bind");
        let port = listener.local_addr().unwrap().port();
        let seen = Arc::new(Mutex::new(Vec::new()));
        let log = seen.clone();
        thread::spawn(move || {
            for stream in listener.incoming() {
                let Ok(s) = stream else { break };
                self.handle(s, &log);
            }
        });
        Running { port, seen }
    }

    fn handle(&self, mut s: TcpStream, log: &Mutex<Vec<String>>) {
        let mut r = BufReader::new(s.try_clone().unwrap());
        let mut line = String::new();
        if r.read_line(&mut line).is_err() {
            return;
        }
        // 读完头,忽略内容(GET 没有正文)
        loop {
            let mut h = String::new();
            if r.read_line(&mut h).unwrap_or(0) == 0 || h.trim().is_empty() {
                break;
            }
        }
        log.lock().unwrap().push(line.trim().to_string());
        let mut parts = line.split_whitespace();
        let method = parts.next().unwrap_or("");
        let target = parts.next().unwrap_or("");
        let (code, body): (u16, Vec<u8>) = if method != "GET" {
            (405, Vec::new())
        } else if let Some(query) = target.strip_prefix("/b?") {
            let prefix = query
                .split('&')
                .find_map(|kv| kv.strip_prefix("prefix="))
                .map(|p| p.replace("%2F", "/"))
                .unwrap_or_default();
            let mut xml = String::from("<ListBucketResult><IsTruncated>false</IsTruncated>");
            for k in self.objects.keys().filter(|k| k.starts_with(&prefix)) {
                xml.push_str(&format!("<Contents><Key>{k}</Key></Contents>"));
            }
            xml.push_str("</ListBucketResult>");
            (200, xml.into_bytes())
        } else if let Some((c, b)) = &self.get_override {
            (*c, b.clone().into_bytes())
        } else {
            let key = target.strip_prefix("/b/").unwrap_or("");
            match self.objects.get(key) {
                Some(v) => (200, v.clone()),
                None => (404, b"<Error><Code>NoSuchKey</Code></Error>".to_vec()),
            }
        };
        let head = format!(
            "HTTP/1.1 {code} X\r\nContent-Length: {}\r\nConnection: close\r\n\r\n",
            body.len()
        );
        let _ = s.write_all(head.as_bytes());
        let _ = s.write_all(&body);
        let _ = s.flush();
    }
}
```

注意:键里含 `-` 与数字、`.mpk`,`url.rs` 的 path 编码对这些不转义(unreserved),所以 `target.strip_prefix("/b/")` 即为原键。若测试里看到 404,先打印 `seen` 核对路径编码。

```rust
//! F346-b:手机从云端拉包。判据在 store(`cloud::latest_key` / `open_cloud_object` /
//! `install_projects_only`),这里只做 IO 编排与类型转换。

fn data_key(v: &[u8]) -> Result<InMemoryKey, FfiError> {

        .try_into()
        .map_err(|_| FfiError::Internal(format!("数据密钥应为 32 字节,实为 {}", v.len())))?;

#[uniffi::export]
pub fn load_cloud_form(cloud_dir: String) -> CloudForm {

/// 读-改-写:只改手机管的字段,桌面专用字段原样保留。
#[uniffi::export]
pub fn save_cloud_form(cloud_dir: String, data_key: Vec<u8>, form: CloudForm) -> Result<(), FfiError> {
    let dir = Path::new(&cloud_dir);
    if is_corrupt(dir) {
        return Err(FfiError::CloudConfigCorrupt);
    }
    let key = self::data_key(&data_key)?;
    let mut cfg = cloud::load(dir);
    cfg.endpoint = form.endpoint.trim().to_string();
    cfg.region = form.region.trim().to_string();
    cfg.bucket = form.bucket.trim().to_string();
    cfg.prefix = form.prefix.trim().to_string();
    cfg.path_style = form.path_style;
    cfg.socks5 = form.socks5.trim().to_string();
    cfg.access_key_id = form.access_key_id.trim().to_string();
    let sk = form.secret_key.trim();
    if !sk.is_empty() {
        set_secret_key_with(&mut cfg, &key, sk)?;
    }
    // 口令不 trim:空格可能就是口令的一部分。
    if !form.passphrase.is_empty() {
        set_passphrase_with(&mut cfg, &key, &form.passphrase)?;
    }
    std::fs::create_dir_all(dir).map_err(|e| FfiError::Storage(e.to_string()))?;
    cloud::save(dir, &cfg)?;
    Ok(())
}

#[uniffi::export]
pub fn reset_cloud_config(cloud_dir: String) -> Result<(), FfiError> {

        Ok(()) => Ok(()),
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
        Err(e) => Err(FfiError::Storage(e.to_string())),
    }
}

/// 拉最新一份并整份替换 `vault_dir`。**全程不打开 `vault_dir`**(只在
/// `install_projects_only` 的 staging 里开),坏库也能被修好(设计 D3)。
/// `now_amz_date` = Kotlin 侧按 UTC 格式化的 `YYYYMMDDTHHMMSSZ`(设计 D7)。
/// 阻塞(网络 + Argon2 + IO),Kotlin 侧放 IO 线程。
#[uniffi::export]
pub fn pull_latest(
    vault_dir: String,
    cloud_dir: String,
    data_key: Vec<u8>,
    now_amz_date: String,
) -> Result<PullSummary, FfiError> {
    let dir = Path::new(&cloud_dir);
    if is_corrupt(dir) {
        return Err(FfiError::CloudConfigCorrupt);
    }
    let key = self::data_key(&data_key)?;
    let cfg = cloud::load(dir);
    // 解不开 ≠ 包坏:不走 `?`,否则落到 `From<StoreError>` 的 BadPack(设计 R2)。
    let sk = secret_key_with(&cfg, &key).map_err(|_| FfiError::CloudConfigUnreadable)?;
    let pass = passphrase_with(&cfg, &key).map_err(|_| FfiError::CloudConfigUnreadable)?;

    let missing: Vec<&str> = [
        ("Endpoint", cfg.endpoint.is_empty()),
        ("Bucket", cfg.bucket.is_empty()),
        ("AK", cfg.access_key_id.is_empty()),
        ("SK", sk.is_empty()),
        ("备份口令", pass.is_empty()),
    ]
    .into_iter()
    .filter_map(|(name, empty)| empty.then_some(name))
    .collect();
    if !missing.is_empty() {
        return Err(FfiError::CloudIncomplete(missing.join("、")));
    }

    let socks5 = (!cfg.socks5.is_empty()).then_some(cfg.socks5.as_str());
    let client = S3Client::new(
        Endpoint {
            base: cfg.endpoint.clone(),
            bucket: cfg.bucket.clone(),
            path_style: cfg.path_style,
        },
        Credentials {
            access_key_id: cfg.access_key_id.clone(),
            secret_access_key: sk,
        },
        cfg.region.clone(),
        socks5,
    )?;
    let keys = client.list_keys(&cfg.prefix, &now_amz_date)?;
    let Some(latest) = latest_key(&cfg.prefix, &keys) else {
        return Err(if keys.is_empty() {
            FfiError::CloudNoBackup(cfg.prefix.clone())
        } else {
            FfiError::CloudNoRecognizedBackup {
                count: keys.len() as u32,
                prefix: cfg.prefix.clone(),
            }
        });
    };
    let blob = match client.get(&latest, &now_amz_date, MAX_OBJECT_BYTES) {
        Err(mullion_cloud::CloudError::Status { code: 404, .. }) => {
            return Err(FfiError::CloudBackupVanished)
        }
        r => r?,
    };
    let text = open_cloud_object(&blob, &pass)?;
    let r = install_projects_only(Path::new(&vault_dir), &text, &pass, &key)?;
    Ok(PullSummary {
        key: latest,
        import: ImportSummary {
            projects: r.projects,
            kept: r.kept,
            dropped: r.dropped,
            broken_nodes: r.broken_nodes,
            unrecognized: r.unrecognized,
        },
    })
}
```

```kotlin
package io.github.kilobitcy.mullion

import android.content.Context
import io.github.kilobitcy.mullion.ffi.CloudForm
import io.github.kilobitcy.mullion.ffi.FfiException
import io.github.kilobitcy.mullion.ffi.loadCloudForm
import io.github.kilobitcy.mullion.ffi.pullLatest
import io.github.kilobitcy.mullion.ffi.saveCloudForm
import java.time.Instant
import java.time.ZoneOffset
import java.time.format.DateTimeFormatter

/** F346-b:云配置存取与拉取。阻塞(网络 + Argon2 + IO),调用方放 IO 线程。 */
object CloudPull {
    private val AMZ = DateTimeFormatter.ofPattern("yyyyMMdd'T'HHmmss'Z'").withZone(ZoneOffset.UTC)

    /** SigV4 的 x-amz-date。手机时间偏差由服务端 RequestTimeTooSkewed 报出(设计 D7)。 */
    fun amzDate(now: Instant = Instant.now()): String = AMZ.format(now)

    const val RESET_NOTE = "本机密钥已失效,旧数据与云配置已清空,请重新填写云配置。\n"

    fun load(ctx: Context): CloudForm = loadCloudForm(VaultKey.cloudDir(ctx).path)

    /** 返回 true = 本次取钥时发生了重置(旧云配置已清,刚存的是重置后的第一份)。 */
    fun save(ctx: Context, form: CloudForm): Boolean {
        val key = VaultKey.load(ctx)
        saveCloudForm(VaultKey.cloudDir(ctx).path, key.dataKey, form)
        return key.reset
    }

    fun pull(ctx: Context, now: Instant = Instant.now()): String {
        val key = VaultKey.load(ctx)
        val s = try {
            pullLatest(VaultKey.vaultDir(ctx).path, VaultKey.cloudDir(ctx).path, key.dataKey, amzDate(now))
        } catch (e: Exception) {
            throw if (key.reset) PackImport.FailedAfterReset(e) else e
        }
        val head = if (key.reset) RESET_NOTE else ""
        return head + "已从云端 ${s.key.substringAfterLast('/')} 更新。" + PackImport.summaryText(s.import)
    }

    fun failureText(e: Exception): String {
        val cause = (e as? PackImport.FailedAfterReset)?.cause ?: e
        val body = when (cause) {
            is FfiException.WrongPassword -> "备份口令不对,请重填后再试"
            is FfiException -> cause.message ?: cause.toString()
            else -> "拉取失败:${cause.message}"
        }
        return (if (e is PackImport.FailedAfterReset) RESET_NOTE else "") + body
    }
}
```

```kotlin
package io.github.kilobitcy.mullion

import android.app.Application
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import io.github.kilobitcy.mullion.ffi.CloudForm
import io.github.kilobitcy.mullion.ffi.resetCloudConfig
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

/** F346-b:云配置表单 + 「从云端更新」。手动拉,不自动(设计 D2)。 */
class CloudViewModel(app: Application) : AndroidViewModel(app) {
    var form by mutableStateOf(CloudPull.load(app))
        private set
    var message by mutableStateOf(if (form.isCorrupt) "云配置损坏,请点「重置云配置」" else "")
        private set
    var busy by mutableStateOf(false)
        private set

    fun edit(f: CloudForm) { form = f }

    fun save() = launchBusy { persist(); message = "已保存" }

    /** 先保存再拉:用户填完直接点更新,不该拿旧配置去拉。 */
    fun pull() = launchBusy {
        persist()
        message = withContext(Dispatchers.IO) { CloudPull.pull(getApplication()) }
    }

    fun reset() = launchBusy {
        withContext(Dispatchers.IO) { resetCloudConfig(VaultKey.cloudDir(getApplication()).path) }
        reload("云配置已重置")
    }

    private suspend fun persist() {
        val reset = withContext(Dispatchers.IO) { CloudPull.save(getApplication(), form) }
        reload(if (reset) CloudPull.RESET_NOTE else "")
    }

    /** 存完重读:SK/口令框清空,只显示「已设置」。 */
    private suspend fun reload(msg: String) {
        form = withContext(Dispatchers.IO) { CloudPull.load(getApplication()) }
        message = msg
    }

    private fun launchBusy(block: suspend () -> Unit) {
        if (busy) return
        busy = true
        viewModelScope.launch {
            try {
                block()
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                message = CloudPull.failureText(e)
            } finally {
                busy = false
            }
        }
    }
}
```

注意:`persist` 失败(如 `CloudConfigCorrupt`)时异常直接冒到 `launchBusy`,`pull` 不会继续 —— 这是想要的。`CloudForm` 是 uniffi 生成的 data class,`copy(...)` 可用。
